import { NextRequest, NextResponse } from 'next/server'

export async function POST(req: NextRequest) {
  const { query, filter } = await req.json()

  // 1. Unsplash Images (ใส่ Key ของคุณใน .env: UNSPLASH_ACCESS_KEY)
  let images: any[] = []
  if (filter === 'all' || filter === 'image') {
    try {
      const unsplashKey = process.env.UNSPLASH_ACCESS_KEY
      if (unsplashKey) {
        const res = await fetch(`https://api.unsplash.com/search/photos?query=${encodeURIComponent(query)}&per_page=3`, {
          headers: { Authorization: `Client-ID ${unsplashKey}` }
        })
        const data = await res.json()
        images = data.results || []
      } else {
        // fallback ถ้าไม่มี key - ใช้ source.unsplash
        images = [{ urls: { regular: `https://source.unsplash.com/800x450/?${encodeURIComponent(query)}` }, alt_description: query }]
      }
    } catch {}
  }

  // 2. YouTube Videos (ใส่ YOUTUBE_API_KEY ใน .env)
  let videos: any[] = []
  if (filter === 'all' || filter === 'video') {
    try {
      const ytKey = process.env.YOUTUBE_API_KEY
      if (ytKey) {
        const res = await fetch(`https://www.googleapis.com/youtube/v3/search?part=snippet&q=${encodeURIComponent(query)}&maxResults=3&type=video&key=${ytKey}`)
        const data = await res.json()
        videos = data.items || []
      }
    } catch {}
  }

  // 3. ElevenLabs Voice (ใส่ ELEVENLABS_API_KEY)
  let audioUrl = null
  if (filter === 'all' || filter === 'audio') {
    audioUrl = "pending-elevenlabs" // ถ้ามี key จะสร้างไฟล์เสียงจริง
  }

  return NextResponse.json({
    query,
    filter,
    timestamp: new Date().toISOString(),
    results: { images, videos, audioUrl, synthesis_note: `สังเคราะห์จาก ${images.length + videos.length} แหล่งสำหรับ "${query}"` }
  })
}

"use client"
import { useEffect, useState } from "react"

type SavedSearch = { id: string; query: string; filter: string; savedAt: string }

export default function PrivateVaultSearch({ currentQuery, currentFilter }: { currentQuery: string, currentFilter: string }) {
  const [items, setItems] = useState<SavedSearch[]>([])

  useEffect(() => {
    const raw = localStorage.getItem("ai-synthesis-vault")
    if (raw) setItems(JSON.parse(raw))
  }, [])

  const save = (list: SavedSearch[]) => {
    setItems(list)
    localStorage.setItem("ai-synthesis-vault", JSON.stringify(list))
  }

  const saveCurrent = () => {
    if (!currentQuery) return
    const item: SavedSearch = { id: Date.now().toString(), query: currentQuery, filter: currentFilter, savedAt: new Date().toISOString() }
    save([item, ...items].slice(0, 20))
  }

  const exportMd = () => {
    const md = items.map(i => `### ${i.query}\n- Filter: ${i.filter}\n- ${new Date(i.savedAt).toLocaleString('th-TH')}\n`).join("\n---\n\n")
    const blob = new Blob([md], { type: "text/markdown" })
    const url = URL.createObjectURL(blob)
    const a = document.createElement("a"); a.href = url; a.download = "my-search-history.md"; a.click()
  }

  return (
    <div className="mt-8 border border-black/10 rounded-[16px] p-5 bg-white">
      <div className="flex justify-between items-center">
        <div className="text-[11px] uppercase tracking-widest text-black/40">Private Vault • {items.length} การสืบค้น</div>
        <div className="flex gap-2">
          <button onClick={saveCurrent} className="text-[11px] px-3 py-1 bg-black text-white rounded-full">บันทึกการค้นหานี้</button>
          <button onClick={exportMd} className="text-[11px] px-3 py-1 border rounded-full">Export .md</button>
        </div>
      </div>
      <div className="mt-4 space-y-2 max-h-[200px] overflow-y-auto">
        {items.length === 0 ? <div className="text-[12px] text-black/30 py-6 text-center">ยังไม่มีบันทึก — กดบันทึกหลังค้นหา</div> :
          items.map(i => (
            <div key={i.id} className="flex justify-between bg-[#FFFEF9] p-3 rounded-[10px]">
              <div><div className="text-[13px]">{i.query}</div><div className="text-[10px] text-black/40">[{i.filter}] • {new Date(i.savedAt).toLocaleTimeString('th-TH')}</div></div>
              <button onClick={() => save(items.filter(x => x.id !== i.id))} className="text-[11px] text-black/20">ลบ</button>
            </div>
          ))
        }
      </div>
    </div>
  )
}

<PrivateVaultSearch currentQuery={query} currentFilter={filter} />
